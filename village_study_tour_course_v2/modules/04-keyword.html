def translate_keywords(keyword_string):
    keywords = keyword_string.split()
    valid_kw = []
    courses = []
    
    for kw in keywords:
        relevance = calculate_relevance(kw)
        if relevance >= MIN_THRESHOLD:
            valid_kw.append(kw)
            courses.extend(match_courses(kw))
    
    return {
        "valid_keywords": valid_kw,
        "courses": courses[:MAX_COURSES]
    }

def calculate_relevance(keyword):
    # 1. 精确匹配预设关键词
    if keyword in KEYWORD_COURSES:
        return KEYWORD_COURSES[keyword]["relevance"]
    
    # 2. 特征匹配打分
    score = 0.0
    for char in keyword:
        if char in RURAL_FEATURES:
            score += 0.12
    
    # 3. 惩罚不相关词
    for pattern in UNRELATED_PATTERNS:
        if pattern in keyword:
            score -= 0.6
    
    return max(0.0, min(1.0, score))
